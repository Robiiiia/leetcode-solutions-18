class Solution { 

    public void rotate(int[][] matrix) { // Метод поворачивает матрицу на месте

        int n = matrix.length; // Получаем количество строк матрицы

        for (int i = 0; i < n; i++) { // Перебираем строки матрицы

            for (int j = i + 1; j < n; j++) { // Перебираем элементы выше главной диагонали

                int temp = matrix[i][j]; // Временно сохраняем текущий элемент

                matrix[i][j] = matrix[j][i]; // Меняем его местами с симметричным элементом

                matrix[j][i] = temp; // Записываем сохранённый элемент на новое место
            }
        }

        for (int i = 0; i < n; i++) { // Перебираем каждую строку матрицы

            int left = 0; // Указываем на первый элемент строки

            int right = n - 1; // Указываем на последний элемент строки

            while (left < right) { // Продолжаем, пока указатели не встретятся

                int temp = matrix[i][left]; // Сохраняем левый элемент

                matrix[i][left] = matrix[i][right]; // Перемещаем правый элемент налево

                matrix[i][right] = temp; // Перемещаем сохранённый элемент направо

                left++; // Сдвигаем левый указатель вправо

                right--; // Сдвигаем правый указатель влево
            }
        }
    }
}
